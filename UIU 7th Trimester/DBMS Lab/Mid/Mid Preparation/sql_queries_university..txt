from student s
join takes t
on s.ID = t.ID
join course c
on t.course_id = c.course_id
join teaches tc
on c.course_id = tc.course_id
join instructor i
on tc.ID = i.ID
where i.name = 'Srinivasan'
group by s.name,s.ID
having count(distinct t.course_id) = (
		select count(c.course_id)
        from course c
        join teaches tc
        on c.course_id = tc.course_id
        join instructor i 
        on tc.ID = i.ID
        where i.name = 'Srinivasan'
